#!/bin/sh
set -e

# Create save path
case "${MWN_SAVE_PATH:=_tmp}" in
  /*) NOTES_PATH="$MWN_SAVE_PATH" ;;
  *) NOTES_PATH="/var/www/html/$MWN_SAVE_PATH" ;;
esac
mkdir -p "$NOTES_PATH"
cp "/var/www/html/notes.htaccess" "$NOTES_PATH/.htaccess"
chown -R www-data:www-data "$NOTES_PATH"

# Optional HTTP basic auth for the whole site.
#   MWN_AUTH_USER + MWN_AUTH_PASS set  -> auth on
#   both empty                          -> auth off
#   MWN_AUTH_REQUIRED=1 and either one empty -> exit (never serve private
#                                               notes without a password)
AUTH_CONF=/etc/apache2/conf-enabled/zz-raw-note-auth.conf
AUTH_FILE=/etc/apache2/raw_note.htpasswd
rm -f "$AUTH_CONF" "$AUTH_FILE"

if [ -n "$MWN_AUTH_USER" ] && [ -n "$MWN_AUTH_PASS" ]; then
    # The password is read from the environment, not passed as an argument,
    # so it never shows in the process list. bcrypt ($2y$) hash.
    php -r '
        $u = getenv("MWN_AUTH_USER");
        if (strpos($u, ":") !== false) { fwrite(STDERR, "MWN_AUTH_USER must not contain \":\"\n"); exit(1); }
        echo $u, ":", password_hash(getenv("MWN_AUTH_PASS"), PASSWORD_BCRYPT), "\n";
    ' > "$AUTH_FILE"
    chown root:www-data "$AUTH_FILE"
    chmod 640 "$AUTH_FILE"
    cat > "$AUTH_CONF" <<EOF
<Directory /var/www/html>
    AuthType Basic
    AuthName "raw_note"
    AuthUserFile $AUTH_FILE
    Require valid-user
</Directory>
EOF
    echo "raw_note: basic auth ON for user '$MWN_AUTH_USER'"
elif [ "$MWN_AUTH_REQUIRED" = "1" ]; then
    echo "raw_note: MWN_AUTH_REQUIRED=1 but MWN_AUTH_USER or MWN_AUTH_PASS is empty. Refusing to start." >&2
    exit 1
fi

# Run default entrypoint
exec docker-php-entrypoint "$@"

</content>

/* token_badge.js
 * Token-count badge for Minimalist Web Notepad.
 *
 * Section A is copied unchanged from raw_token_counter.js: it defines the
 * badge's look. To take a style change from a newer raw_token_counter.js,
 * copy its badge block over Section A. Nothing in Section B depends on it
 * except setBadge().
 *
 * Section B fetches counts for the SAVED note from GET /<note>?tokens=1
 * (handled by index.php):
 *   - once when the page opens
 *   - after every successful save ('note-saved' event fired by script.js)
 * The note text is never sent; the tokenizer reads the saved file.
 *
 * Badge values:  number = count,  "—" = note not saved yet / empty,
 *                "too big" = note over the tokenizer's size limit,
 *                "err 502" etc. = server answered with that HTTP status
 *                  (502 = tokenizer container down or unreachable),
 *                "err net" = request never got an answer: network down,
 *                  timeout, or blocked inside the browser by an extension,
 *                "err json" = HTTP 200 but the body was not JSON.
 * The next save retries in every error case. Details go to the console.
 *
 * index.php passes the URL as a data attribute on this script tag:
 * data-tokens-url.
 */
(function () {
  'use strict';

  const me = document.currentScript;
  const TOKENS_URL = me && me.dataset.tokensUrl;
  if (!TOKENS_URL) return;

  // ===== Section A: badge (copied unchanged from raw_token_counter.js) =====

  const badge = document.createElement('div');
  badge.id = 'tc-token-badge';

  const num1 = document.createElement('span');
  const num2 = document.createElement('span');

  num1.id = 'tc-token-o';
  num2.id = 'tc-token-anthropic';

  badge.appendChild(num1);
  badge.appendChild(num2);

  Object.assign(badge.style, {
    position: 'fixed',
    top: '11%',
    right: '0px',
    transform: 'translateY(-50%)',
    zIndex: '999999',
    pointerEvents: 'none',
    display: 'flex',
    flexDirection: 'column',
    alignItems: 'stretch',
    whiteSpace: 'nowrap',
    overflow: 'hidden',
    padding: '0',
    borderRadius: '15px 0 0 15px',
    boxShadow: '0 3px 10px rgba(0,0,0,0.08)',
    fontSize: '19px',
    fontWeight: 'bold',
    fontFamily: 'sans-serif'
  });

  Object.assign(num1.style, {
    background: '#7048e8',
    color: '#ffffff',
    paddingTop: '1.5px',
    paddingBottom: '0px',
    paddingLeft: '8px',
    paddingRight: '4px'
  });

  Object.assign(num2.style, {
    background: '#ae3ec9',
    color: '#ffffff',
    paddingTop: '0px',
    paddingBottom: '0px',
    paddingLeft: '8px',
    paddingRight: '4px'
  });

  num1.textContent = '—';
  num2.textContent = '—';

  // Appended to <html>, NOT <body>: styles.css has a "body * { ... !important }"
  // rule that would override the badge's font, weight and line height.
  document.documentElement.appendChild(badge);

  // Hide badge while the mouse is over it
  document.addEventListener('mousemove', (e) => {
    const r = badge.getBoundingClientRect();
    const overBadge =
      e.clientX >= r.left &&
      e.clientX <= r.right &&
      e.clientY >= r.top &&
      e.clientY <= r.bottom;
    badge.style.opacity = overBadge ? '0' : '1';
  });

  function setBadge(oText, aText) {
    num1.textContent = oText;
    num2.textContent = aText;
  }

  // ===== Section B: counts for the saved note =====

  // Must be longer than index.php's 30 s curl timeout.
  const TIMEOUT_MS = 35000;

  let inFlight = false; // a count request is running
  let pending = false;  // a save happened while busy or hidden: fetch once more

  function fmt(v) {
    return typeof v === 'number' ? v.toLocaleString('en-US') : '?';
  }

  function refresh() {
    if (inFlight || document.hidden) {
      pending = true;
      return;
    }
    inFlight = true;
    pending = false;

    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), TIMEOUT_MS);

    fetch(TOKENS_URL, {
      cache: 'no-store',
      signal: ctrl.signal
    })
      .then((res) => {
        if (res.status === 404) { setBadge('—', '—'); return; }
        if (res.status === 413) { setBadge('too big', 'too big'); return; }
        if (!res.ok) {
          return res.text().then((t) => {
            console.warn('[token_badge] HTTP ' + res.status, t);
            setBadge('err ' + res.status, 'err ' + res.status);
          });
        }
        return res.json().then((j) => setBadge(fmt(j.o), fmt(j.anthropic)));
      })
      .catch((e) => {
        if (e && e.name === 'SyntaxError') {
          // HTTP 200 but the body was not JSON.
          console.warn('[token_badge] bad JSON from ' + TOKENS_URL, e);
          setBadge('err json', 'err json');
          return;
        }
        // fetch() itself failed: no HTTP answer reached the page.
        console.warn('[token_badge] no answer from ' + TOKENS_URL, e);
        setBadge('err net', 'err net');
      })
      .finally(() => {
        clearTimeout(timer);
        inFlight = false;
        if (pending) refresh();
      });
  }

  window.addEventListener('note-saved', refresh);
  document.addEventListener('visibilitychange', () => {
    if (!document.hidden && pending) refresh();
  });

  refresh();
})();

</content>

      - default

      - /tmp

    cpus: 1         # cap at 1 core (a limit, nothing is reserved); works on 1-core hosts
    restart: unless-stopped

networks:
  tok:
    internal: true

/*! Minimalist Web Notepad | https://github.com/pereorga/minimalist-web-notepad
 *  Optimized save loop: event-driven, debounced, raw (gzip for big notes),
 *  only marks "saved" on HTTP 2xx, warns before closing with unsaved changes.
 *
 *  Text access goes through `ed`: the CodeMirror editor from editor.js when
 *  it loaded (window.rawNoteEditor), otherwise the plain textarea.
 */
(function () {
    'use strict';

    var IDLE_MS    = 600;        // save this long after the last keystroke / paste
    var RETRY_MIN  = 2000;       // first retry delay after a failed save
    var RETRY_MAX  = 30000;      // retry delay cap
    var GZIP_MIN   = 32 * 1024;  // gzip bodies at or above this many characters

    var textarea  = document.getElementById('content');
    var printable = document.getElementById('printable');
    var ed = window.rawNoteEditor || {
        getValue: function () { return textarea.value; },
        onChange: function (fn) { textarea.addEventListener('input', fn); },
        focus: function () { textarea.focus({ preventScroll: true }); }
    };
    var baseTitle = document.title;
    var canGzip   = typeof CompressionStream === 'function';

    var dirty = false;   // textarea changed since the last successful save started
    var saving = false;  // a request is in flight
    var timer = null;
    var retryDelay = RETRY_MIN;

    // Title prefix shows state:  "• note" = unsaved,  "⚠ note" = save failing.
    // The title is written only when the state changes, not on every key.
    var shownState = '';
    function setState(prefix) {
        if (prefix === shownState) return;
        shownState = prefix;
        document.title = prefix ? prefix + ' ' + baseTitle : baseTitle;
    }

    function schedule(ms) {
        clearTimeout(timer);
        timer = setTimeout(save, ms);
    }

    // Only a change event marks the note dirty: the full text is read only
    // when saving, never on each keystroke.
    ed.onChange(function () {
        dirty = true;
        setState('•');
        if (!saving) schedule(IDLE_MS);
    });

    function buildBody(text) {
        if (canGzip && text.length >= GZIP_MIN) {
            var gz = new Blob([text]).stream().pipeThrough(new CompressionStream('gzip'));
            return new Response(gz).blob().then(function (blob) {
                return { body: blob, gzip: true };
            });
        }
        return Promise.resolve({ body: text, gzip: false });
    }

    function save() {
        if (!dirty || saving) return;
        saving = true;
        dirty = false;
        var text = ed.getValue();

        buildBody(text).then(function (b) {
            var headers = { 'Content-Type': 'text/plain; charset=UTF-8' };
            if (b.gzip) headers['X-Content-Gzip'] = '1';
            return fetch(window.location.href, {
                method: 'POST',
                headers: headers,
                body: b.body,
                cache: 'no-store'
            });
        }).then(function (res) {
            if (!res.ok) throw new Error('HTTP ' + res.status);
            saving = false;
            retryDelay = RETRY_MIN;
            // Tell token_badge.js the saved file changed (it re-counts it).
            window.dispatchEvent(new CustomEvent('note-saved'));
            if (dirty) {
                schedule(IDLE_MS);   // user typed during the request
            } else {
                setState('');
            }
        }).catch(function () {
            saving = false;
            dirty = true;            // keep the change; try again later
            setState('⚠');
            schedule(retryDelay);
            retryDelay = Math.min(retryDelay * 2, RETRY_MAX);
        });
    }

    // Warn before leaving with unsaved changes.
    window.addEventListener('beforeunload', function (e) {
        if (dirty || saving) {
            e.preventDefault();
            e.returnValue = '';
        }
    });

    // Fill the print copy only when printing, instead of keeping a second
    // full copy of the note in the DOM at all times.
    window.addEventListener('beforeprint', function () {
        printable.textContent = ed.getValue();
    });
    window.addEventListener('afterprint', function () {
        printable.textContent = '';
    });

    ed.focus();
})();

</content>
